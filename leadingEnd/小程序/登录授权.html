登录授权 ​

微信登录流程 ​

小程序登录流程分为三步：

  1. 调用 wx.login() 获取临时 code
  2. 将 code 传给后端，后端调用 code2Session 接口换取 openid 和 session_key
  3. 后端返回自定义登录态 token，前端存储并用后续请求
js
App({
  onLaunch() {
    this.login()
  },
  login() {
    wx.login({
      success: (res) => {
        if (res.code) {
          // 将 code 发送到后端
          wx.request({
            url: 'https://api.example.com/login',
            method: 'POST',
            data: { code: res.code },
            success: (response) => {
              // 存储 token
              wx.setStorageSync('token', response.data.token)
            }
          })
        } else {
          console.error('登录失败：', res.errMsg)
        }
      }
    })
  }
})

登录态检查 ​

每次启动时检查 token 是否有效，无效则重新登录。

js
const TOKEN_EXPIRY = 7200 // token 有效期（秒）

function checkLogin() {
  const token = wx.getStorageSync('token')
  const loginTime = wx.getStorageSync('loginTime')

  if (!token || !loginTime) return false

  const now = Date.now()
  const elapsed = (now - loginTime) / 1000

  return elapsed < TOKEN_EXPIRY
}

完整登录流程示例 ​

js
// utils/auth.js
const TOKEN_KEY = 'token'

export function login() {
  return new Promise((resolve, reject) => {
    // 1. 检查已有 token
    const token = wx.getStorageSync(TOKEN_KEY)
    if (token) {
      return resolve(token)
    }

    // 2. 获取 code
    wx.login({
      success: (res) => {
        if (!res.code) {
          return reject(new Error('登录失败'))
        }

        // 3. 后端换取登录态
        wx.request({
          url: 'https://api.example.com/login',
          method: 'POST',
          data: { code: res.code },
          success: (response) => {
            const { token } = response.data
            wx.setStorageSync(TOKEN_KEY, token)
            resolve(token)
          },
          fail: reject
        })
      },
      fail: reject
    })
  })
}

注意事项 ​

  • wx.login() 每次调用都会生成新的 code，一个 code 只能使用一次
  • session_key 不应传到前端或通过前端转发，应直接在后端使用
  • 不要将 openid 和 session_key 泄露给第三方
  • 建议为每个请求添加 token 有效性校验机制
