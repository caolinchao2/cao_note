python
import re

text = "联系方式: alice@example.com 和 bob@test.org"

# re.match — 从开头匹配，不匹配则返回 None
m = re.match(r"联系方式", text)
print(m.group() if m else "无匹配")  # 联系方式

# re.search — 搜索整个字符串，返回第一个匹配
m = re.search(r"\w+@\w+\.\w+", text)
print(m.group())  # alice@example.com

# re.findall — 返回所有匹配的列表
emails = re.findall(r"\w+@\w+\.\w+", text)
print(emails)  # ['alice@example.com', 'bob@test.org']

# re.finditer — 返回迭代器，每个元素是 Match 对象
for match in re.finditer(r"\w+@\w+\.\w+", text):
    print(match.group(), match.span())

python
import re

text = "2024-12-25 和 2025-01-01"

pattern = r"(?P<year>\d{4})-(?P<month>\d{2})-(?P<day>\d{2})"

for m in re.finditer(pattern, text):
    print(m.group())       # 完整匹配 2024-12-25
    print(m.group(1))      # 2024
    print(m.groups())      # ('2024', '12', '25')
    print(m.groupdict())   # {'year': '2024', 'month': '12', 'day': '25'}

import re

text = "联系方式: alice@example.com"

# re.sub — 替换匹配内容
result = re.sub(r"\w+@\w+\.\w+", "***@***.***", text)
print(result)  # 联系方式: ***@***.***

# 使用函数动态替换
def mask_email(match):
    name = match.group().split("@")[0]
    return f"{name[0]}***@{match.group().split('@')[1]}"

result = re.sub(r"\w+@\w+\.\w+", mask_email, text)
print(result)  # 联系方式: a***@example.com

import re

text = "<div>内容1</div><div>内容2</div>"

greedy = re.search(r"<div>.*</div>", text)

non_greedy = re.search(r"<div>.*?</div>", text)

import json

data = {
    "name": "张三",
    "age": 30,
    "skills": ["Python", "Docker", "K8s"],
    "active": True,
    "salary": None,
}

json_str = json.dumps(data, indent=2, ensure_ascii=False)
print(json_str)

parsed = json.loads(json_input)
print(parsed["name"])  # 李四

# 文件读写
with open("data.json", "w", encoding="utf-8") as f:

with open("data.json", "r", encoding="utf-8") as f:

    print(loaded["name"])  # 张三

# 自定义 JSON 序列化（处理 datetime 等类型）
from datetime import datetime

def json_serializer(obj):
    if isinstance(obj, datetime):
        return obj.isoformat()
    raise TypeError(f"Type {type(obj)} not serializable")

now = datetime.now()
data = {"event": "meeting", "time": now}
print(json.dumps(data, default=json_serializer))
# {"event": "meeting", "time": "2025-01-15T14:30:00"}

import csv

# reader — 读取 CSV
with open("data.csv", "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    header = next(reader)  # 跳过表头
    for row in reader:
        print(row)  # 每行是一个列表

# writer — 写入 CSV
with open("output.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.writer(f)
    writer.writerow(["姓名", "年龄", "城市"])      # 写表头
    writer.writerow(["张三", 30, "北京"])
    writer.writerow(["李四", 25, "上海"])
    writer.writerows([                       # 批量写入
        ["王五", 35, "广州"],
        ["赵六", 28, "深圳"],
    ])

# DictReader — 按字典读取
with open("output.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(row["姓名"], row["年龄"])  # 按列名访问

# DictWriter — 按字典写入
with open("dict_output.csv", "w", newline="", encoding="utf-8") as f:
    fieldnames = ["姓名", "年龄", "城市"]

from collections import deque

dq = deque(["a", "b", "c"], maxlen=5)  # maxlen 可选

limited = deque(maxlen=3)
