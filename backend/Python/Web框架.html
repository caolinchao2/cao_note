class UserForm(forms.ModelForm):
    class Meta:
        model = User
        fields = ('name', 'email', 'age')  # 表单包含的字段

            'name': forms.TextInput(attrs={'class': 'form-control'}),

        }

# myapp/views.py
def create_user(request):
    if request.method == 'POST':
        form = UserForm(request.POST)
        if form.is_valid():
            form.save()  # 保存到数据库
            return HttpResponse("User created successfully!")
    else:
        form = UserForm()
    return render(request, 'create_user.html', {'form': form})

from flask import Flask
from flask_sqlalchemy import SQLAlchemy

app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    
    def __repr__(self):
        return f'<User {self.username}>'

# 创建数据库
with app.app_context():
    db.create_all()

# 使用模型
@app.route('/add_user/<username>/<email>')
def add_user(username, email):
    user = User(username=username, email=email)
    db.session.add(user)
    db.session.commit()
    return f'User {username} added'

@app.route('/users')
def get_users():
    users = User.query.all()
    return ', '.join([user.username for user in users])

class Order(BaseModel):
    items: list[str]
    total: float

    @field_validator("total")
    @classmethod
    def check_total(cls, v: float) -> float:
        if v < 0:
            raise ValueError("金额不能为负")
        return v

    @model_validator(mode="after")
    def check_items(self) -> "Order":

class PaginationParams(BaseModel):
    page: int = Field(default=1, ge=1)
    page_size: int = Field(default=10, ge=1, le=100)

class PaginatedResult(BaseModel):
    items: list
    total: int
    page: int
    page_size: int
    total_pages: int

def paginate(query, total: int, params: PaginationParams) -> PaginatedResult:
    return PaginatedResult(
        items=list(query),
        total=total,
        page=params.page,
        page_size=params.page_size,
        total_pages=ceil(total / params.page_size),

框架选择建议 ​

  1. Django：

    • 适合需要快速开发大型Web应用的团队
    • 内置功能丰富，减少第三方依赖
    • 适合需要Admin后台的项目

  2. Flask：

    • 适合小型项目或API服务
    • 适合学习Web开发的初学者
    • 适合需要高度定制化的项目

  3. FastAPI：

    • 适合构建高性能的RESTful API
    • 适合需要异步支持的项目
    • 适合微服务架构
    • 适合需要自动文档的项目

部署建议 ​

  1. Django：

    • WSGI服务器：Gunicorn + Nginx
    • 容器化：Docker + Kubernetes
    • 云平台：AWS Elastic Beanstalk, Google App Engine

  2. Flask：

    • WSGI服务器：Gunicorn + Nginx
    • 容器化：Docker + Kubernetes
    • 云平台：Heroku, AWS Lambda

  3. FastAPI：

    • ASGI服务器：Uvicorn + Gunicorn + Nginx
    • 容器化：Docker + Kubernetes
    • 云平台：AWS Lambda, Google Cloud Functions

总结 ​

Python提供了多种优秀的Web框架，选择合适的框架取决于项目的需求、团队的经验和项目的规模。无论选择哪种框架，都应该遵循最佳实践，编写清晰、可维护的代码，并确保应用的安全性和性能。

  • Django适合需要全功能框架的大型项目
  • Flask适合需要灵活性的小型项目和API服务
  • FastAPI适合需要高性能和自动文档的现代API开发

通过学习和掌握这些框架，您可以构建各种类型的Web应用，从简单的API到复杂的企业级应用。
