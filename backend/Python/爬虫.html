Python 爬虫 ​

网络爬虫与数据采集技术，从基础请求到框架应用。

爬虫基础概念 ​

什么是爬虫 ​

网络爬虫（Web Crawler）是一种自动化程序，通过模拟浏览器发送 HTTP 请求，获取网页内容并从中提取所需数据。

爬虫的分类 ​

  • 通用爬虫：搜索引擎的核心，爬取整个互联网（如 Google、百度）
  • 聚焦爬虫：针对特定主题进行爬取
  • 增量爬虫：只爬取新增或更新的内容
  • 深网爬虫：爬取需要表单提交或登录后才能访问的页面

# my_spider/spiders/quotes.py
import scrapy

class QuotesSpider(scrapy.Spider):
    name = "quotes"
    allowed_domains = ["quotes.toscrape.com"]
    start_urls = ["https://quotes.toscrape.com/"]

    def parse(self, response):
        for quote in response.css("div.quote"):
            yield {
                "text": quote.css("span.text::text").get(),
                "author": quote.css("small.author::text").get(),
                "tags": quote.css("div.tags a.tag::text").getall(),
            }
        # 翻页
        next_page = response.css("li.next a::attr(href)").get()
        if next_page:
            yield response.follow(next_page, callback=self.parse)

Item 与 Pipeline ​

python
# items.py
import scrapy
class QuoteItem(scrapy.Item):
    text = scrapy.Field()
    author = scrapy.Field()
    tags = scrapy.Field()

# pipelines.py
import json
class JsonPipeline:
    def open_spider(self, spider):
        self.file = open("quotes.json", "w", encoding="utf-8")
        self.file.write("[")
        self.first = True

    def close_spider(self, spider):
        self.file.write("]")
        self.file.close()

    def process_item(self, item, spider):
        line = json.dumps(dict(item), ensure_ascii=False)
        self.file.write(f"{',' if not self.first else ''}{line}")
        self.first = False
        return item

# settings.py 中启用

requests + BeautifulSoup ​

python
import requests
from bs4 import BeautifulSoup
import csv
import time
import random

def fetch_page(url, session):
    try:
        resp = session.get(url, timeout=10)
        resp.raise_for_status()
    except requests.RequestException:
        return [], None
    soup = BeautifulSoup(resp.text, "lxml")
    quotes = []
    for q in soup.select("div.quote"):
        quotes.append({
            "text": q.select_one("span.text").get_text(strip=True),
            "author": q.select_one("small.author").get_text(strip=True),
            "tags": ", ".join(t.get_text(strip=True) for t in q.select("a.tag")),
        })
    btn = soup.select_one("li.next a")
    next_url = "http://quotes.toscrape.com" + btn["href"] if btn else None
    return quotes, next_url

session = requests.Session()
session.headers.update({"User-Agent": "Mozilla/5.0 Chrome/120.0.0.0"})
url = "http://quotes.toscrape.com/page/1/"
all_quotes = []
for page in range(1, 6):
    if not url: break
    print(f"第 {page} 页...")
    quotes, url = fetch_page(url, session)
    all_quotes.extend(quotes)
    time.sleep(random.uniform(0.5, 1.5))

with open("quotes.csv", "w", newline="", encoding="utf-8-sig") as f:
    w = csv.DictWriter(f, fieldnames=["text", "author", "tags"])
    w.writeheader(); w.writerows(all_quotes)
print(f"完成！共 {len(all_quotes)} 条")

python
import asyncio, csv
from playwright.async_api import async_playwright

async def main():
    all_quotes = []
    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True)
        page = await browser.new_page()
        url = "https://quotes.toscrape.com/js/"
        for i in range(1, 6):
            await page.goto(url, wait_until="networkidle")
            await page.wait_for_selector("div.quote")
            for q in await page.query_selector_all("div.quote"):
                all_quotes.append({
                    "text": await q.query_selector("span.text").inner_text(),
                    "author": await q.query_selector("small.author").inner_text(),
                    "tags": ", ".join([await t.inner_text() for t in await q.query_selector_all("a.tag")]),
                })
            btn = await page.query_selector("li.next a")
            url = "https://quotes.toscrape.com/js/" + await btn.get_attribute("href") if btn else None
            if not url: break
            await asyncio.sleep(1)
        await browser.close()
    with open("quotes_js.csv", "w", newline="", encoding="utf-8-sig") as f:
        w = csv.DictWriter(f, fieldnames=["text", "author", "tags"])
        w.writeheader(); w.writerows(all_quotes)
    print(f"完成！共 {len(all_quotes)} 条")
asyncio.run(main())

反爬虫策略（最佳实践） ​

python
import time, random
from functools import wraps

# 频率控制器
class RateLimiter:
    def __init__(self, cps=2):
        self.min_interval = 1.0 / cps
        self.last_call = 0.0
    def wait(self):
        elapsed = time.time() - self.last_call
        if elapsed < self.min_interval:
            time.sleep(self.min_interval - elapsed)
        self.last_call = time.time()

# 重试装饰器
def retry(max_retries=3, delay=1, backoff=2):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            _d = delay
            for i in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if i == max_retries - 1:
                        raise
                    print(f"重试 {i+1}: {e}")
                    time.sleep(_d)
                    _d *= backoff
        return wrapper
    return decorator

总结 ​

技术选型 ​

场景推荐方案
静态页面requests + BeautifulSoup
异步高并发httpx / aiohttp
大型爬虫项目Scrapy
JS 动态渲染Playwright
分布式爬取Scrapy + Scrapy-Redis

核心原则 ​

  1. 合规优先：遵守 robots.txt 和相关法律法规
  2. 频率控制：合理控制请求速度，不造成服务器负担
  3. 异常处理：完善的错误处理和重试机制
  4. 数据质量：清洗和验证爬取的数据
  5. 效率平衡：在速度和稳定性之间找到平衡
