1. PostgreSQL 概述 ​

1.1 特点 ​

PostgreSQL 是一个功能强大的开源对象关系型数据库系统，拥有超过 30 年的活跃开发历史。其主要特点包括：

  • ACID 事务：完整支持原子性、一致性、隔离性、持久性
  • 扩展性：支持自定义数据类型、操作符、索引方法、函数
  • SQL 标准：高度兼容 SQL:2016 标准
  • MVCC：多版本并发控制，读写互不阻塞
  • 丰富的数据类型：JSON/JSONB、数组、范围类型、网络类型等
  • 索引类型丰富：B-tree、Hash、GiST、GIN、BRIN、SP-GiST
  • 逻辑复制与流复制：支持高可用与读写分离
  • 外部数据包装器（FDW）：可访问其他数据库、文件、Web 服务

1.3 架构 ​

进程模型 ​

PostgreSQL 使用 进程模型（而非线程模型）：

Postmaster（主进程）
  ├── 后台进程
  │   ├── WAL Writer      — 写 WAL 日志
  │   ├── Checkpointer    — 执行检查点
  │   ├── Autovacuum Launcher — 启动自动清理
  │   ├── WAL Archiver    — WAL 归档
  │   ├── Stats Collector — 统计信息收集
  │   └── Logger          — 日志记录
  └── 客户端连接进程
      ├── postgres: user1 database1
      ├── postgres: user2 database2
      └── ...

每个客户端连接对应一个独立的操作系统进程，进程间通过共享内存通信。

共享内存 ​

  • shared_buffers：数据页缓存，默认 128MB
  • WAL Buffer：WAL 日志缓冲区，默认 16MB
  • 锁空间：存储进程间同步所需的各种锁

WAL（Write-Ahead Logging） ​

WAL 是 PostgreSQL 保证数据持久性的核心机制：

  1. 数据修改先写入 WAL 日志，再写入数据文件
  2. 崩溃恢复时回放 WAL 日志，恢复到一致状态
  3. 支持 PITR（时间点恢复）和流复制

2.5 JSON / JSONB ​

PostgreSQL 提供了两种 JSON 类型：

类型存储索引处理速度
JSON文本（保留原格式）不支持直接索引慢（每次需解析）
JSONB二进制（去除空格、键有序）支持 GIN 索引快（无需解析）
sql
CREATE TABLE books (
    id SERIAL PRIMARY KEY,
    data JSONB
);

INSERT INTO books (data) VALUES
    ('{"title": "PostgreSQL 指南", "author": {"name": "Tom", "age": 35}, "tags": ["database", "sql"], "price": 59.9}'),
    ('{"title": "SQL 入门", "author": {"name": "Jerry", "age": 28}, "tags": ["sql"], "price": 39.9}');

-- JSONB 查询操作符
SELECT data->>'title' AS title FROM books;                          -- -> 返回 JSON，->> 返回文本
SELECT data->'author'->>'name' AS author_name FROM books;           -- 嵌套路径
SELECT data #>> '{author, name}' AS author_name FROM books;         -- #>> 使用路径表达式

-- JSONB 存在查询
SELECT * FROM books WHERE data @> '{"tags": ["sql"]}';              -- 包含（@>）
SELECT * FROM books WHERE data ? 'price';                           -- 键是否存在（?）
SELECT * FROM books WHERE data ?| ARRAY['price', 'rating'];         -- 任一键存在（?|）
SELECT * FROM books WHERE data ?& ARRAY['title', 'author'];         -- 所有键存在（?&）

-- JSONB 创建索引
CREATE INDEX idx_books_data ON books USING GIN (data);
CREATE INDEX idx_books_title ON books USING GIN ((data->>'title')); -- 表达式索引

-- JSONB 修改
UPDATE books SET data = jsonb_set(data, '{price}', '69.9', false) WHERE data->>'title' = 'PostgreSQL 指南';
UPDATE books SET data = data || '{"rating": 4.5}'::jsonb;          -- 合并新字段
UPDATE books SET data = data - 'price';                            -- 删除字段

CREATE OR REPLACE FUNCTION safe_transfer(from_account INTEGER, to_account INTEGER, amount NUMERIC)
RETURNS TEXT AS $$
BEGIN
    -- 检查余额
    IF (SELECT balance FROM accounts WHERE id = from_account FOR UPDATE) < amount THEN
        RAISE EXCEPTION '余额不足, 当前余额: %', (SELECT balance FROM accounts WHERE id = from_account);
    END IF;

    UPDATE accounts SET balance = balance - amount WHERE id = from_account;
    UPDATE accounts SET balance = balance + amount WHERE id = to_account;

    RETURN '转账成功';
EXCEPTION
    WHEN UNIQUE_VIOLATION THEN
        RETURN '转账失败: 唯一约束冲突';
    WHEN OTHERS THEN
        RETURN '转账失败: ' || SQLERRM;  -- SQLERRM 获取错误消息
END;
$$ LANGUAGE plpgsql;
CREATE OR REPLACE FUNCTION safe_transfer(from_account INTEGER, to_account INTEGER, amount NUMERIC)
RETURNS TEXT AS $$
BEGIN
    -- 检查余额
    IF (SELECT balance FROM accounts WHERE id = from_account FOR UPDATE) < amount THEN
        RAISE EXCEPTION '余额不足, 当前余额: %', (SELECT balance FROM accounts WHERE id = from_account);
    END IF;

    UPDATE accounts SET balance = balance - amount WHERE id = from_account;
    UPDATE accounts SET balance = balance + amount WHERE id = to_account;

    RETURN '转账成功';
EXCEPTION
    WHEN UNIQUE_VIOLATION THEN
        RETURN '转账失败: 唯一约束冲突';
    WHEN OTHERS THEN
        RETURN '转账失败: ' || SQLERRM;  -- SQLERRM 获取错误消息
END;
$$ LANGUAGE plpgsql;

10.2 pgvector（向量相似度搜索） ​

pgvector 为 PostgreSQL 添加向量存储和相似度搜索能力，广泛应用于 AI/LLM 应用中的语义搜索。

sql
-- 安装
CREATE EXTENSION vector;

-- 创建向量表
CREATE TABLE embeddings (
    id SERIAL PRIMARY KEY,
    content TEXT,
    embedding VECTOR(1536)  -- OpenAI text-embedding-ada-002 维度
);

-- 插入向量数据
INSERT INTO embeddings (content, embedding) VALUES
    ('PostgreSQL 是一款强大的数据库', '[0.1, 0.2, ...]'::VECTOR(1536)),
    ('向量搜索用于语义匹配', '[0.3, 0.1, ...]'::VECTOR(1536));

-- 相似度搜索

-- L2 距离（欧几里得距离）
SELECT content, embedding <-> '[0.15, 0.18, ...]'::VECTOR(1536) AS distance
FROM embeddings
ORDER BY distance
LIMIT 5;

-- 内积（最大内积搜索）
SELECT content, embedding <#> '[0.15, 0.18, ...]'::VECTOR(1536) AS inner_product
FROM embeddings
ORDER BY inner_product
LIMIT 5;

-- 余弦相似度
SELECT content, 1 - (embedding <=> '[0.15, 0.18, ...]'::VECTOR(1536)) AS cosine_similarity
FROM embeddings
ORDER BY cosine_similarity DESC
LIMIT 5;

-- 创建索引加速搜索
-- IVFFlat 索引（需要先有数据来训练索引）
CREATE INDEX ON embeddings USING IVFFLAT (embedding vector_cosine_ops) WITH (lists = 100);
-- IVFFlat 索引选项：
--   vector_l2_ops: L2 距离
--   vector_ip_ops: 内积
--   vector_cosine_ops: 余弦相似度

-- HNSW 索引（PG16+, pgvector 0.5+）
CREATE INDEX ON embeddings USING HNSW (embedding vector_cosine_ops) WITH (m = 16, ef_construction = 200);

10.4 TimescaleDB（时序数据） ​

TimescaleDB 是 PostgreSQL 的时序数据扩展，自动管理基于时间的分区。

sql
-- 安装（需要先安装 TimescaleDB 软件包）
CREATE EXTENSION IF NOT EXISTS timescaledb;

-- 创建时序表
CREATE TABLE sensor_data (
    time TIMESTAMPTZ NOT NULL,
    sensor_id INTEGER,
    temperature FLOAT,
    humidity FLOAT,
    pressure FLOAT
);

-- 转换为超表（自动按时间分区）
SELECT create_hypertable('sensor_data', 'time');

-- 可选：按 sensor_id 进一步分区
SELECT create_hypertable('sensor_data', 'time', chunk_time_interval => INTERVAL '1 day');
-- 或追加空间分区（PostgreSQL 分区键）
ALTER TABLE sensor_data SET (
    timescaledb.compress,
    timescaledb.compress_segmentby = 'sensor_id'
);

-- 插入数据
INSERT INTO sensor_data (time, sensor_id, temperature, humidity)
SELECT
    generate_series(NOW() - INTERVAL '30 days', NOW(), INTERVAL '1 minute'),
    1,
    20 + random() * 15,
    50 + random() * 30;

-- 时序查询
-- 时间桶聚合
SELECT
    time_bucket('1 hour', time) AS hour,
    sensor_id,
    AVG(temperature) AS avg_temp,
    MAX(temperature) AS max_temp,
    MIN(temperature) AS min_temp
FROM sensor_data
WHERE time > NOW() - INTERVAL '7 days'
GROUP BY hour, sensor_id
ORDER BY hour;

-- 连续聚合（自动物化视图）
CREATE MATERIALIZED VIEW sensor_hourly
WITH (timescaledb.continuous) AS
SELECT
    time_bucket('1 hour', time) AS bucket,
    sensor_id,
    AVG(temperature) AS avg_temp,
    MAX(temperature) AS max_temp,
    MIN(temperature) AS min_temp
FROM sensor_data
GROUP BY bucket, sensor_id;

-- 数据保留策略
SELECT add_retention_policy('sensor_data', INTERVAL '90 days');  -- 自动删除 90 天前的数据

-- 压缩
ALTER TABLE sensor_data SET (
    timescaledb.compress,
    timescaledb.compress_segmentby = 'sensor_id',
    timescaledb.compress_orderby = 'time DESC'
);
SELECT add_compression_policy('sensor_data', INTERVAL '7 days');  -- 7 天前的数据自动压缩
